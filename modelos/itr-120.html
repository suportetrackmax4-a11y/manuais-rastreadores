J16

APN,trackmax.br,max,max#
SZCS#APN=trackmax.br#USERPPP=max#PWPPP=max
SZCS#FREQ=60#PULSE=3600#SOURCE_OFF_TYPE=1#ACC_OFF_FREQ=3600
SZCS#GT06SEL=1#GT06IEXVOL=2#ACCLT=2#GPS_DISSLP=0#MTK_DISSLP=0#Slpdisconnect=1#sleepupdate=1
SZCS#ANGLEVALUE=05-030#VIBCHK=1:250#VIBL=6
TIMER,60,3600#
HBT,3600,3600#
ParâmetroSignificado provável
APN= / USERPPP= / PWPPP=APN, usuário e senha PPP
FREQ=60Intervalo de envio com ignição ligada: 60 s
PULSE=3600Heartbeat: 3600 s
SOURCE_OFF_TYPE=1Bloqueio imediato (ver pasta J14 2g)
ACC_OFF_FREQ=3600Intervalo com ignição desligada: 3600 s
GT06SEL=1Seleciona o protocolo GT06
GT06IEXVOL=2Envio da tensão externa no protocolo GT06
ACCLT=2Modo de detecção da ACC
GPS_DISSLP=0 / MTK_DISSLP=0Permite o sleep do GPS e do módulo
Slpdisconnect=1Desconecta no sleep
sleepupdate=1Envia atualização durante o sleep
ANGLEVALUE=05-030Envio por ângulo (05 / 030 graus)
VIBCHK=1:250 / VIBL=6Checagem e nível de vibração
TIMER,60,3600#60 s ligado / 3600 s desligado
HBT,3600,3600#Heartbeat

LV12

TIMER,60,300#
HBT,5#
STATIC,3600#
  • TIMER,60,300#: 60 s com ignição ligada, 300 s com ignição desligada
  • HBT,5#: heartbeat de 5 min
  • STATIC,3600#: envio a cada 3600 s quando parado
